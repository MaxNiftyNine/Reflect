cmake_minimum_required(VERSION 3.22)
if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
project(reflect_viewer LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "Reflect's viewer requires macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
add_executable(reflect-viewer MACOSX_BUNDLE viewer.mm)
target_include_directories(reflect-viewer PRIVATE ../protocol)
target_compile_options(reflect-viewer PRIVATE -fobjc-arc -Wall -Wextra)
target_link_libraries(reflect-viewer PRIVATE "-framework Cocoa" "-framework Metal" "-framework MetalKit" "-framework QuartzCore")
set_target_properties(reflect-viewer PROPERTIES OSX_ARCHITECTURES arm64 OUTPUT_NAME Reflect MACOSX_BUNDLE_BUNDLE_NAME Reflect MACOSX_BUNDLE_GUI_IDENTIFIER dev.reflect.viewer MACOSX_BUNDLE_BUNDLE_VERSION 0.1 MACOSX_BUNDLE_SHORT_VERSION_STRING 0.1)
